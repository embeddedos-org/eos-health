cmake_minimum_required(VERSION 3.20)
project(eos_shared_constants C)

# Host-only suite. Nothing here links Zephyr, so it runs on a bare ubuntu
# runner with gcc and nothing else.
enable_testing()

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_FLAGS "-Wall -Wextra")

# Must be normalised: an unresolved ".../shared/tests/.." prefix would make
# every globbed path match the "/tests/" exclusion below and silently scan
# nothing.
get_filename_component(SHARED_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)

# ─── Regression guard for #29 ─────────────────────────────────────────────────
# firmware/shared/provisioning/provisioning.c defined PROV_MAGIC as
# 0xEA5P0001U. 'P' is not a hex digit -- it introduces the binary exponent of a
# hexadecimal *floating* constant, and 'U' is not a valid floating suffix, so
# every translation unit expanding the macro failed to compile. It survived
# because the only build target listing provisioning.c
# (firmware/build-system) stops at a FATAL_ERROR on the unmet nRF5 SDK long
# before a compiler runs, and because the repo had no build workflow (#27).
#
# A malformed preprocessing number is only diagnosed where the macro is
# *expanded*, never at the #define itself. So this guard scrapes every
# hexadecimal object-like macro out of the real firmware/shared sources and
# forces each one through the compiler in a real expansion context. No Zephyr
# headers are involved, which is what lets it run in CI when the firmware
# itself cannot be built.
file(GLOB_RECURSE SHARED_C_SOURCES
    ${SHARED_ROOT}/*.c
    ${SHARED_ROOT}/*.h
)

set(GEN_BODY "/* Generated by firmware/shared/tests/CMakeLists.txt. Do not edit. */\n")
string(APPEND GEN_BODY "#include <stdint.h>\n\n")

set(CONST_INDEX 0)
set(SCANNED_FILES 0)
foreach(src IN LISTS SHARED_C_SOURCES)
    # Don't scan the tests themselves.
    if(src MATCHES "/tests/")
        continue()
    endif()
    math(EXPR SCANNED_FILES "${SCANNED_FILES} + 1")

    file(STRINGS ${src} HEX_DEFINES
         REGEX "^[ \t]*#[ \t]*define[ \t]+[A-Za-z_][A-Za-z0-9_]*[ \t]+0[xX][0-9A-Za-z]+[ \t]*$")

    foreach(line IN LISTS HEX_DEFINES)
        string(REGEX MATCH "([A-Za-z_][A-Za-z0-9_]*)[ \t]+(0[xX][0-9A-Za-z]+)" _m "${line}")
        set(macro_name  ${CMAKE_MATCH_1})
        set(macro_value ${CMAKE_MATCH_2})

        file(RELATIVE_PATH rel ${SHARED_ROOT} ${src})
        string(APPEND GEN_BODY "/* ${rel}: ${macro_name} */\n")
        string(APPEND GEN_BODY "#undef ${macro_name}\n")
        string(APPEND GEN_BODY "#define ${macro_name} ${macro_value}\n")
        # The expansion is the point: this is the line that fails to compile
        # if the literal is malformed.
        string(APPEND GEN_BODY
               "const unsigned long long eos_const_${CONST_INDEX} = (unsigned long long)(${macro_name});\n\n")
        math(EXPR CONST_INDEX "${CONST_INDEX} + 1")
    endforeach()
endforeach()

if(CONST_INDEX EQUAL 0)
    message(FATAL_ERROR
        "constant scan matched 0 macros across ${SCANNED_FILES} files in ${SHARED_ROOT}. "
        "The regex or the layout changed; this suite would otherwise pass by testing nothing.")
endif()
message(STATUS "shared constants guard: ${CONST_INDEX} hex macros from ${SCANNED_FILES} files")

# Pin the provisioning magic itself. 0xEA50 is the EoS namespace prefix used
# across this firmware (ble_manager.c exposes 0xEA500001..0xEA500008 as the EoS
# Custom Service UUIDs) and 0001 is the record version.
string(APPEND GEN_BODY "_Static_assert(PROV_MAGIC == 0xEA500001U, \"PROV_MAGIC changed value\");\n")
string(APPEND GEN_BODY "_Static_assert(CRASH_LOG_MAGIC == 0xEA5C0DE1U, \"CRASH_LOG_MAGIC changed value\");\n")
string(APPEND GEN_BODY "_Static_assert(DATA_BUF_MAGIC == 0xDA7AB0FFU, \"DATA_BUF_MAGIC changed value\");\n\n")
string(APPEND GEN_BODY "int main(void) { return 0; }\n")

set(GEN_FILE ${CMAKE_CURRENT_BINARY_DIR}/test_shared_constants.c)
file(GENERATE OUTPUT ${GEN_FILE} CONTENT "${GEN_BODY}")

add_executable(test_shared_constants ${GEN_FILE})
add_test(NAME test_shared_constants COMMAND test_shared_constants)
